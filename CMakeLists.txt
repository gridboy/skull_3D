cmake_minimum_required(VERSION 3.16)
project(skull LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 REQUIRED)
add_library(renderer src/volume_renderer.cpp)
target_include_directories(renderer PUBLIC include)
add_executable(skull src/main.cpp)
target_compile_definitions(skull PRIVATE SKULL_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data/cthead")
target_link_libraries(skull PRIVATE renderer glfw OpenGL::GL)
if(APPLE)
    find_library(COREGRAPHICS_FRAMEWORK CoreGraphics REQUIRED)
    find_library(COREFOUNDATION_FRAMEWORK CoreFoundation REQUIRED)
    find_library(CORETEXT_FRAMEWORK CoreText REQUIRED)
    target_link_libraries(skull PRIVATE ${COREGRAPHICS_FRAMEWORK}
        ${COREFOUNDATION_FRAMEWORK} ${CORETEXT_FRAMEWORK})
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(renderer_test tests/renderer_test.cpp)
    target_link_libraries(renderer_test PRIVATE renderer)
    # Keep assertions active in Release builds as well.
    target_compile_options(renderer_test PRIVATE -UNDEBUG)
    add_test(NAME renderer COMMAND renderer_test)
    set_tests_properties(renderer PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
